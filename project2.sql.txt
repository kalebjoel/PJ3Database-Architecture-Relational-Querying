DESCRIBE film_actor;
DESCRIBE payment;
DESCRIBE staff;

-- Display customers and their addresses in Texas ordered by last name
SELECT a.address_id, a.phone, c.customer_id, c.last_name
FROM ADDRESS a
INNER JOIN CUSTOMER c
	ON a.address_id = c.address_id
WHERE a.district = 'Texas'
ORDER BY c.last_name;

-- Display the actors who appeared in film 7, ordered by last name
SELECT a.actor_id, a.first_name, a.last_name, f.film_id
FROM ACTOR a
INNER JOIN FILM_ACTOR f
	ON a.actor_id = f.actor_id
WHERE f.film_id = 7
ORDER BY a.last_name;

-- Display cities located in the United States, ordered alphabetically
SELECT ci.city_id, ci.city, co.country_id, co.country
FROM CITY ci
INNER JOIN COUNTRY co
	ON ci.country_id = co.country_id
WHERE co.country = 'United States'
ORDER BY ci.city
LIMIT 10;

-- Display all actors whos last names start with P and any films they have ordered
by last name
SELECT a.actor_id, a.first_name, a.last_name, fa.film_id, f.title
FROM ACTOR a
LEFT JOIN FILM_ACTOR fa
	ON a.actor_id = fa.actor_id
LEFT JOIN FILM f
	ON fa.film_id = f.film_id
WHERE a.last_name LIKE 'P%'
ORDER BY a.last_name
LIMIT 10;

-- Display customers with payments greater than 11, ordered by last name
SELECT c.customer_id, c.first_name, c.last_name, p.payment_id, p.amount
FROM CUSTOMER c
RIGHT JOIN PAYMENT p
	ON c.customer_id = p.customer_id
WHERE p.amount > 11
ORDER BY c.last_name;

-- Display films and their associated languages, ordered by language
SELECT f.film_id, f.title, f.original_language_id, l.language_id, l.name
FROM FILM f
LEFT JOIN LANGUAGE l
	ON f.language_id = l.language_idWHERE f.film_id > 0
ORDER BY l.name
LIMIT 10;

-- Display pairs of addresses located in the same city
SELECT a1.address_id AS address_1, a2.address_id AS address_2, a1.city_id
FROM ADDRESS a1
JOIN ADDRESS a2
	ON a1.city_id = a2.city_id
WHERE a1.address_id < a2.address_id
ORDER BY a1.city_id
LIMIT 10;

-- Display movies that have the exact same runtime ordered by runtime
SELECT f1.length, f1.title as movie_1, f2.title as movie_2
FROM FILM f1
JOIN FILM f2
	ON f1.length = f2.length
WHERE f1.film_id < f2.film_id
ORDER BY f1.length
LIMIT 10;
